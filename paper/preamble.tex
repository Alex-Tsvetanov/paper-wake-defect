% No en or em dashes. The cite package joins a compressed citation range with --,
% which would print as an en dash. A hyphen instead.
\renewcommand{\citedash}{-}

% IEEEtran.bst reads this control entry when it is cited first. article does not
% define the command. Repeated author names then stay names, rather than a dash.
\makeatletter
\newcommand{\bstctlcite}[1]{%
  \@bsphack
  \@for\@citeb:=#1\do{%
    \edef\@citeb{\expandafter\@firstofone\@citeb}%
    \if@filesw\immediate\write\@auxout{\string\citation{\@citeb}}\fi}%
  \@esphack}
\makeatother
